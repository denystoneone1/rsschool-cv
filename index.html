Code Example

Full Backup Script:

#!/usr/bin/env bash

set -euo pipefail

# =========================
# CONFIG
# =========================
SOURCE="/home"
DEST="/backup"
LOG_FILE="/var/log/backup.log"
RETENTION_DAYS=7

# Exclusions (edit as needed)
EXCLUDES=(
    "/proc"
    "/sys"
    "/dev"
    "/tmp"
    "/run"
    "/mnt"
    "/media"
    "/lost+found"
)

# =========================
# INIT
# =========================
TIMESTAMP=$(date +"%Y-%m-%d_%H-%M-%S")
BACKUP_DIR="$DEST/$TIMESTAMP"
LATEST_LINK="$DEST/latest"

mkdir -p "$DEST"
mkdir -p "$BACKUP_DIR"

echo "[$(date)] Starting backup..." >> "$LOG_FILE"

# =========================
# BUILD EXCLUDE PARAMS
# =========================
RSYNC_EXCLUDES=()
for EX in "${EXCLUDES[@]}"; do
    RSYNC_EXCLUDES+=(--exclude="$EX")
done

# =========================
# RUN RSYNC (incremental)
# =========================
if [ -d "$LATEST_LINK" ]; then
    LINK_DEST="--link-dest=$LATEST_LINK"
else
    LINK_DEST=""
fi

rsync -aAXv \
    --delete \
    "${RSYNC_EXCLUDES[@]}" \
    $LINK_DEST \
    "$SOURCE/" "$BACKUP_DIR/" >> "$LOG_FILE" 2>&1

# =========================
# UPDATE SYMLINK
# =========================
rm -f "$LATEST_LINK"
ln -s "$BACKUP_DIR" "$LATEST_LINK"

# =========================
# CLEAN OLD BACKUPS
# =========================
find "$DEST" -maxdepth 1 -type d -mtime +$RETENTION_DAYS -exec rm -rf {} \; >> "$LOG_FILE" 2>&1 || true

echo "[$(date)] Backup completed: $BACKUP_DIR" >> "$LOG_FILE"
